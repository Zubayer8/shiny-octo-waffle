            const jobsGrid = document.getElementById('jobsGrid');

            const noResultsMessage = document.getElementById('no-results-message');

            const jobSectionTitle = document.getElementById('job-section-title');

            const testimonialSlider = document.getElementById('testimonialSlider');

            const prevBtn = document.getElementById('prevTestimonial');

            const nextBtn = document.getElementById('nextTestimonial');

            

            // --- JOB CARD GENERATION ---

            function generateJobCards(filterCategory = 'all') {

                jobsGrid.innerHTML = ''; 

                const filteredJobs = filterCategory === 'all' 

                    ? jobListings 

                    : jobListings.filter(job => job.category === filterCategory);


                if (filteredJobs.length === 0) {

                    jobsGrid.style.display = 'none';

                    noResultsMessage.style.display = 'block';

                } else {

                    jobsGrid.style.display = 'grid';

                    noResultsMessage.style.display = 'none';

                    

                    filteredJobs.forEach((job, index) => {

                        const jobCard = document.createElement('div');

                        jobCard.className = 'job-card';

                        jobCard.innerHTML = `

                            <div class="job-image">

                                <img src="${job.image}" alt="${job.title}" loading="lazy">

                                <div class="job-badge">New</div>

                            </div>

                            <div class="job-content">

                                <h3 class="job-title">${job.title}</h3>

                                <span class="job-type">${job.type}</span>

                                <p class="job-description">Explore opportunities in this field with competitive compensation.</p>

                                <a href="${job.link}" target="_blank" rel="noopener noreferrer" class="apply-btn">Apply Now <i class="fas fa-arrow-right"></i></a>

                            </div>`;

                        jobsGrid.appendChild(jobCard);

                        setTimeout(() => { jobCard.classList.add('animate'); }, 100 * (index % 10));

                    });

                }

            }


            // --- FILTERING LOGIC ---

            const allFilterLinks = document.querySelectorAll('.nav-filter');

            

            function updateActiveFilter(category) {

                // Remove active class from all filter links

                allFilterLinks.forEach(link => link.classList.remove('active-filter'));

                

                // Add active class to the selected category links

                document.querySelectorAll(`.nav-filter[data-category="${category}"]`).forEach(el => el.classList.add('active-filter'));

                

                // Handle special case for dropdown parent

                const dropdownParent = document.querySelector('.dropdown > a');

                if (category !== 'all' && document.querySelector(`.dropdown .nav-filter[data-category="${category}"]`)) {

                    dropdownParent.classList.add('active-filter');

                } else {

                    dropdownParent.classList.remove('active-filter');

                }

            }


            allFilterLinks.forEach(link => {

                link.addEventListener('click', function(e) {

                    // This is the key fix for footer links

                    e.preventDefault(); 

                    

                    const category = this.dataset.category;

                    

                    updateActiveFilter(category);

                    

                    jobSectionTitle.textContent = category === 'all' ? 'Featured Opportunities' : `Opportunities in ${category}`;

                    generateJobCards(category);

                    

                    // Scroll to job section only if the link is in the header

                    const isHeaderLink = this.closest('header');

                    if(isHeaderLink) { 

                         document.getElementById('job-section').scrollIntoView({ behavior: 'smooth' });

                    }

                });

            });


            // --- TESTIMONIAL SLIDER ---

            let currentSlide = 0;

            function generateTestimonials() {

                if(!testimonialSlider) return;

                testimonialSlider.innerHTML = '';

                testimonials.forEach(testimonial => {

                    const testimonialEl = document.createElement('div');

                    testimonialEl.className = 'testimonial';

                    testimonialEl.innerHTML = `

                        <div class="testimonial-img"><img src="${testimonial.image}" alt="${testimonial.name}" loading="lazy"></div>

                        <p class="testimonial-text">"${testimonial.text}"</p>

                        <h4 class="testimonial-name">${testimonial.name}</h4>

                        <p class="testimonial-position">${testimonial.position}</p>`;

                    testimonialSlider.appendChild(testimonialEl);

                });

            }


            function showSlide(index) {

                if(testimonialSlider) {

                    testimonialSlider.style.transform = `translateX(-${index * 100}%)`;

                    currentSlide = index;

                }

            }

            if(prevBtn && nextBtn) {

                prevBtn.addEventListener('click', () => {

                    const newIndex = (currentSlide > 0) ? currentSlide - 1 : testimonials.length - 1;

                    showSlide(newIndex);

                });


                nextBtn.addEventListener('click', () => {

                    const newIndex = (currentSlide < testimonials.length - 1) ? currentSlide + 1 : 0;

                    showSlide(newIndex);

                });

                

                // Auto-slide

                setInterval(() => {

                    nextBtn.click();

                }, 5000);

            }

            

            // --- SMOOTH SCROLL FOR OTHER ANCHOR LINKS ---

            document.querySelectorAll('a[href^="#"]').forEach(anchor => {

                anchor.addEventListener('click', function (e) {

                    // Prevent this for filter links, as they are handled separately

                    if (this.classList.contains('nav-filter')) return;


                    const targetEl = document.querySelector(this.getAttribute('href'));

                    if (targetEl) {

                        e.preventDefault();

                        targetEl.scrollIntoView({ behavior: 'smooth' });

                    }

                });

            });

            

            // --- INITIAL PAGE LOAD ---

            generateJobCards();

            generateTestimonials();

        });
